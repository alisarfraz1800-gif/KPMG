  method EVENTID_F4SET_GET_ENTITYSET.

 data: lo_facade    type ref to /iwbep/if_mgw_dp_int_facade.

    data: lv_top           type string,
          lv_skip          type i,
          lv_search_string type string,
          lv_EVENT         type  string,
          r_EVENT          type range of zrfq_header-event_no,
          l_EVENT          like line of r_EVENT.

**-- get request header object reference
    lo_facade ?= /iwbep/if_mgw_conv_srv_runtime~get_dp_facade( ).

**- get number of records requested
    clear: lv_top,lv_skip.
    lv_top = io_tech_request_context->get_top( ).

**- get number of lines that should be skipped
    lv_skip = io_tech_request_context->get_skip( ).
    lv_top = lv_top + lv_skip.

    if iv_search_string is not initial.
      lv_EVENT  = iv_search_string.
      lv_EVENT = |%{ lv_EVENT }%|.
      select distinct EVENT_NO as EVENTid  from ZRFQ_HEADER as a where event_no like @lv_EVENT
                AND RFQWTPR = 'X'
        order by event_no
        into table @et_entityset.
    else.
      select distinct EVENT_NO as EVENTid from ZRFQ_HEADER as a where
       RFQWTPR = 'X'
        order by event_no
        into table @et_entityset.
    endif.

**- skip/top logic to restrict no of records
    if lv_skip is not initial and lv_skip <> '0'.
      delete et_entityset from 1 to lv_skip.
    endif.


  endmethod.

        where ekgrp like @lv_ekgrp or eknam like @lv_ekgrp.
    else.
      select ekgrp, eknam from t024 into table @lt_t024
        where ekgrp <> @space.
    endif.

    if lt_t024 is not initial.
      select distinct ekgrp from eban into table @data(lt_eban_grp)
        where ekgrp <> @space.
      loop at lt_t024 into data(ls_t024).
        if line_exists( lt_eban_grp[ ekgrp = ls_t024-ekgrp ] ).
          append value #( purchasegroup = ls_t024-ekgrp
                          text          = ls_t024-eknam ) to et_entityset.
        endif.
      endloop.
      sort et_entityset by purchasegroup text.
    endif.


**- Skip/Top Logic to Restrict No Of Records
    if lv_skip is not initial and lv_skip <> '0'.
      delete et_entityset from 1 to lv_skip.
    endif.
  endmethod.
